1. Organization Schema - סכמת ארגון

קריטי

יש להוסיף את קוד ה-JSON-LD הבא לתוך תג <head> בכל דפי האתר. הסכמה כוללת פרטי עסק מלאים, נקודות קשר, אזור שירות, רשתות חברתיות וסטטוס נציגות מותגים.

ⓘ מיקום: בתוך <head> לפני תג </head> הסוגר. מומלץ להוסיף באמצעות Rank Math או ישירות ב-header.php של התבנית.

2. Canonical Tags - תגיות קנוניקל

קריטי

יש להוסיף תגית canonical בכל עמוד באתר בתוך ה-<head>. כל עמוד מפנה לעצמו (self-referencing) כדי למנוע תוכן כפול.

עמוד הבית

HTML Homepage 
<link rel="canonical" href="https://www.laguna-water.co.il/" />

עמודי קטגוריה - מוצרים

HTML Product Category Pages 
<!-- קטגוריית ברי מים -->
<link rel="canonical" href="https://www.laguna-water.co.il/product-category/water-bars/" />

<!-- קטגוריית מטהרי מים -->
<link rel="canonical" href="https://www.laguna-water.co.il/product-category/water-purifiers/" />

<!-- קטגוריית כיורים -->
<link rel="canonical" href="https://www.laguna-water.co.il/product-category/kitchen-sinks/" />

<!-- קטגוריית מטחנות אשפה -->
<link rel="canonical" href="https://www.laguna-water.co.il/product-category/garbage-disposals/" />

<!-- קטגוריית פילטרים -->
<link rel="canonical" href="https://www.laguna-water.co.il/product-category/filters/" />

עמודי מוצר בודד

HTML Individual Product Pages (Self-referencing) 
<!-- כל עמוד מוצר מפנה לעצמו -->
<!-- דוגמה: -->
<link rel="canonical" href="https://www.laguna-water.co.il/product/[product-slug]/" />

עמודי תוכן ובלוג

HTML Content / Blog Pages (Self-referencing) 
<!-- כל עמוד תוכן מפנה לעצמו -->
<!-- דוגמה: עמוד אודות -->
<link rel="canonical" href="https://www.laguna-water.co.il/about/" />

<!-- דוגמה: עמוד צור קשר -->
<link rel="canonical" href="https://www.laguna-water.co.il/contact/" />

<!-- דוגמה: פוסט בלוג -->
<link rel="canonical" href="https://www.laguna-water.co.il/blog/[post-slug]/" />
ⓘ אם Rank Math או Yoast מותקנים - יש לוודא שהתוסף מייצר את הקנוניקל אוטומטית. אם כן, אין צורך להוסיף ידנית. יש רק לבדוק שהכתובות נכונות ומכילות www.
⚠ חשוב: לוודא עקביות - כל הקנוניקלים עם www ועם לוכסן (/) סוגר. לא לערבב בין www ו-non-www.

JavaScript FAQ Accordion Script 
// Laguna FAQ Accordion
document.addEventListener('DOMContentLoaded', function() {
  const faqItems = document.querySelectorAll('.laguna-faq-item');

  faqItems.forEach(function(item) {
    const button = item.querySelector('.laguna-faq-question');

    button.addEventListener('click', function() {
      const isActive = item.classList.contains('active');

      // Close all items
      faqItems.forEach(function(otherItem) {
        otherItem.classList.remove('active');
        otherItem.querySelector('.laguna-faq-question')
          .setAttribute('aria-expanded', 'false');
      });

      // Toggle clicked item
      if (!isActive) {
        item.classList.add('active');
        button.setAttribute('aria-expanded', 'true');
      }
    });
  });
});

<script>
function loadGoogleReviews() {
  const service = new google.maps.places.PlacesService(
    document.createElement('div')
  );

  // Replace with Laguna's actual Place ID
  service.getDetails({
    placeId: 'LAGUNA_PLACE_ID', // Get from Google Maps
    fields: ['reviews', 'rating', 'user_ratings_total']
  }, function(place, status) {
    if (status === google.maps.places.PlacesServiceStatus.OK) {
      // Display rating
      document.getElementById('review-score').textContent = place.rating;
      document.getElementById('review-count').textContent =
        '(' + place.user_ratings_total + ' ביקורות)';
      document.getElementById('review-stars').innerHTML =
        renderStars(place.rating);

      // Display reviews (max 5)
      const grid = document.getElementById('reviews-grid');
      place.reviews.slice(0, 5).forEach(function(review) {
        grid.innerHTML += `
          <div class="laguna-review-card">
            <div class="laguna-review-header">
              <img src="${review.profile_photo_url}"
                   alt="${review.author_name}"
                   class="laguna-review-avatar" />
              <div>
                <strong>${review.author_name}</strong>
                <div class="laguna-review-stars">
                  ${renderStars(review.rating)}
                </div>
              </div>
            </div>
            <p>${review.text}</p>
          </div>
        `;
      });
    }
  });
}

function renderStars(rating) {
  let stars = '';
  for (let i = 1; i <= 5; i++) {
    stars += i <= rating
      ? '<span style="color:#f59e0b">★</span>'
      : '<span style="color:#d1d5db">★</span>';
  }
  return stars;
}

loadGoogleReviews();
</script>
