שורש התקלה

ערך של שדה טקסט עשיר מוזרק ישירות אל תוך מחרוזת JSON. הערך מכיל ירידות שורה גולמיות, ואלה אינן חוקיות בתוך מחרוזת JSON ופוסלות את הבלוק כולו. כך נראה הפלט בפועל:

"description": "


החזון של מכללת כרכור הוא להיות בית ומקום בטוח להתפתחות, כדי להעמיד דור של מטפלים
מקצועיים, קשובים ופתוחים ...

",

שגיאות נוספות שגוגל מדווח על אותם בלוקים

  • uploadDate בפורמט לא תקין וללא אזור זמן.
  • חסר contentUrl או embedUrl ברוב הסרטונים.
  • בחלק מהבלוקים חסרים name או description.

התיקון

לבנות את הסכמה כמערך PHP ולהצפין אותו פעם אחת באמצעות wp_json_encode. הפונקציה מטפלת אוטומטית בירידות שורה, במרכאות ובתווים בעברית, ומונעת את התקלה מלכתחילה.

<?php
/**
 * תיקון סכמת VideoObject
 * הסיבה לתקלה: ערך שדה טקסט עשיר מודפס ישירות בתוך מחרוזת JSON,
 * וירידות השורה שבתוכו שוברות את המבנה. הפתרון: לבנות מערך ולהצפין אותו פעם אחת.
 */

// לפני (שגוי) - הדפסה גולמית בתוך מרכאות:
// "description": "<?php echo $video_description; ?>",

// אחרי (תקין):
$video = array(
    '@context'     => 'https://schema.org',
    '@type'        => 'VideoObject',
    'name'         => $video_title,
    'description'  => wp_strip_all_tags( $video_description ),
    'thumbnailUrl' => $video_thumb_url,                 // חובה לתוצאה עשירה
    'uploadDate'   => get_the_date( 'c', get_the_ID() ), // ISO-8601 עם אזור זמן
    'embedUrl'     => $video_embed_url,                  // או contentUrl
);

echo '<script type="application/ld+json">'
   . wp_json_encode( $video, JSON_UNESCAPED_UNICODE | JSON_UNESCAPED_SLASHES )
   . '</script>';

אותו תיקון פותר את כל 55 הבלוקים, מכיוון שכולם מיוצרים מאותה תבנית.

function karkur_print_schema( $data ) {
    if ( empty( $data ) ) { return; }
    echo '<script type="application/ld+json">'
        . wp_json_encode( $data, JSON_UNESCAPED_UNICODE | JSON_UNESCAPED_SLASHES )
        . '</script>' . PHP_EOL;
}

/** 2. ProfilePage - עמוד מרצה */
function karkur_schema_profile() {
    if ( ! is_singular( 'teachers' ) ) { return; }
    $id     = get_the_ID();
    $person = array(

/** 3. Course - עמודי קורסים ותוכניות */
function karkur_schema_course() {
    if ( ! is_singular( array( 'courses', 'plans' ) ) ) { return; }
    $id   = get_the_ID();
    $desc = get_the_excerpt( $id );
    if ( ! $desc ) {
        $desc = wp_trim_words( wp_strip_all_tags( get_post_field( 'post_content', $id ) ), 40 );
    }
    karkur_print_schema( array(
        '@context'    => 'https://schema.org',
        '@type'       => 'Course',
        '@id'         => get_permalink( $id ) . '#course',
        'url'         => get_permalink( $id ),
        'name'        => get_the_title( $id ),
        'description' => $desc,
        'provider'    => array( '@id' => KARKUR_ORG_ID ),
    ) );
}
add_action( 'wp_head', 'karkur_schema_course', 20 );

/** 4. BreadcrumbList - עמודים פנימיים */
function karkur_schema_breadcrumbs() {
    if ( is_front_page() ) { return; }
    $items = array(
        array( '@type' => 'ListItem', 'position' => 1, 'name' => 'בית', 'item' => home_url( '/' ) ),
    );
    $map = array(
        'courses'  => array( 'קורסים',              home_url( '/8/' ) ),
        'plans'    => array( 'תוכניות רב שנתיות',   home_url( '/allplans/' ) ),
        'teachers' => array( 'סגל המרצים',          home_url( '/teachers/' ) ),
    );
    $pt = get_post_type();
    if ( is_singular() && isset( $map[ $pt ] ) ) {
        $items[] = array( '@type' => 'ListItem', 'position' => 2,
                          'name' => $map[ $pt ][0], 'item' => $map[ $pt ][1] );
        $items[] = array( '@type' => 'ListItem', 'position' => 3, 'name' => get_the_title() );
    } elseif ( is_singular( 'post' ) ) {
        $cats = get_the_category();
        if ( ! empty( $cats ) ) {
            $items[] = array( '@type' => 'ListItem', 'position' => 2,
                              'name' => $cats[0]->name,
                              'item' => get_category_link( $cats[0]->term_id ) );
        }
        $items[] = array( '@type' => 'ListItem', 'position' => count( $items ) + 1,
                          'name' => get_the_title() );
    } else {
        return;
    }
    karkur_print_schema( array(
        '@context'        => 'https://schema.org',
        '@type'           => 'BreadcrumbList',
        'itemListElement' => $items,
    ) );
}
add_action( 'wp_head', 'karkur_schema_breadcrumbs', 20 );

/** 5. CollectionPage - עמוד רשימת המרצים */
function karkur_schema_teachers_list() {
    if ( ! is_post_type_archive( 'teachers' ) && ! is_page( 'teachers' ) ) { return; }
    $q = new WP_Query( array(
        'post_type'      => 'teachers',
        'posts_per_page' => -1,
        'orderby'        => 'menu_order title',
        'order'          => 'ASC',
        'no_found_rows'  => true,
    ) );
    $items = array();
    $i     = 0;
    foreach ( $q->posts as $p ) {
        $i++;
        $items[] = array(
            '@type'    => 'ListItem',
            'position' => $i,
            'item'     => array(
                '@type' => 'Person',
                'name'  => get_the_title( $p ),
                'url'   => get_permalink( $p ),
            ),
        );
    }
    wp_reset_postdata();
    karkur_print_schema( array(
        '@context'   => 'https://schema.org',
        '@type'      => 'CollectionPage',
        '@id'        => home_url( '/teachers/#collectionpage' ),
        'url'        => home_url( '/teachers/' ),
        'name'       => 'סגל המרצים - מכללת כרכור',
        'mainEntity' => array(
            '@type'           => 'ItemList',
            'numberOfItems'   => $i,
            'itemListElement' => $items,
        ),
    ) );
}
add_action( 'wp_head', 'karkur_schema_teachers_list', 20 );
