// components/JobPostingSchema.tsx
import Head from 'next/head';

const REGION_TO_CITY: Record<string, {locality: string; region: string}> = {
  'גוש דן': { locality: 'Tel Aviv', region: 'Center District' },
  'השרון': { locality: 'Netanya', region: 'Sharon District' },
  'חיפה והצפון': { locality: 'Haifa', region: 'Northern District' },
  'הדרום': { locality: "Be'er Sheva", region: 'Southern District' },
  'השפלה': { locality: 'Rehovot', region: 'Central District' },
  'ירושלים והסביבה': { locality: 'Jerusalem', region: 'Jerusalem District' },
  'אילת והערבה': { locality: 'Eilat', region: 'Southern District' },
};

const SCOPE_TO_TYPE: Record<string, string> = {
  'משרה מלאה': 'FULL_TIME',
  'משרה חלקית': 'PART_TIME',
  'משרה זמנית': 'TEMPORARY',
  'משרת סטודנט': 'INTERN',
  'משמרות': 'OTHER',
  'עבודה מועדפת': 'OTHER',
  'משרה היברידית/עבודה מהבית': 'FULL_TIME', // + TELECOMMUTE added below
  'משרה פנים ארגונית': 'OTHER',
};

export default function JobPostingSchema({ job, datePosted }: { job: any; datePosted: string }) {
  if (job.jobStatuses?.nodes?.[0]?.name !== 'published') return null;

  const cleanTitle = job.title.replace(/^\d+K\s*/, '');

  let description = (job.jobFields.jobdescription || '')
    .replace(/<article[^>]*>/g, '').replace(/<\/article>/g, '');
  if (job.jobFields.jobrequirements) description += job.jobFields.jobrequirements;
  if (job.jobFields.jobskills) description += `<p>${job.jobFields.jobskills}</p>`;

  const locations = (job.jobAreas?.nodes || []).map((area: any) => {
    const mapped = REGION_TO_CITY[area.name] || { locality: area.name, region: area.name };
    return {
      '@type': 'Place',
      address: {
        '@type': 'PostalAddress',
        addressLocality: mapped.locality,
        addressRegion: mapped.region,
        addressCountry: 'IL',
      },
    };
  });

  const employmentTypes = (job.jobScopes?.nodes || [])
    .map((s: any) => SCOPE_TO_TYPE[s.name] || 'OTHER');
  // Add TELECOMMUTE for hybrid scope OR description keywords
  if (job.jobScopes?.nodes?.some((s: any) => s.name === 'משרה היברידית/עבודה מהבית')) {
    employmentTypes.push('TELECOMMUTE');
  }
  const descText = (job.jobFields.jobdescription || '');
  if (!employmentTypes.includes('TELECOMMUTE') && (descText.includes('עבודה מהבית') || descText.includes('היברידי'))) {
    employmentTypes.push('TELECOMMUTE');
  }

  const parentProf = job.jobProfessions?.nodes?.find((p: any) => p.parentDatabaseId === null);
  const childProf = job.jobProfessions?.nodes?.find((p: any) => p.parentDatabaseId !== null);

  const validDate = new Date(datePosted);
  validDate.setDate(validDate.getDate() + 60);
  const validThrough = validDate.toISOString().split('T')[0] + 'T23:59:59+03:00';

  const schema: any = {
    '@context': 'https://schema.org/',
    '@type': 'JobPosting',
    title: cleanTitle,
    description,
    identifier: { '@type': 'PropertyValue', name: 'Manpower Israel', value: job.jobFields.jobcode },
    datePosted,
    validThrough,
    employmentType: employmentTypes.length === 1 ? employmentTypes[0] : employmentTypes,
    directApply: true,
    hiringOrganization: {
      '@type': 'Organization',
      name: 'Manpower Israel',
      alternateName: 'מנפאואר ישראל',
      sameAs: 'https://www.manpower.co.il/',
      logo: 'https://manpowercontent.manpower.co.il/wp-content/uploads/2023/05/footer-manpower-logo.svg_.png',
    },
    jobLocation: locations.length === 1 ? locations[0] : locations,
  };

  if (parentProf) schema.industry = parentProf.name;
  if (childProf) schema.occupationalCategory = childProf.name;
  if (job.jobExperiences?.nodes?.[0]) schema.experienceRequirements = job.jobExperiences.nodes[0].name;

  return (
    <Head>
      <script type="application/ld+json" dangerouslySetInnerHTML={{ __html: JSON.stringify(schema) }} />
    </Head>
  );
}

// components/ServiceSchema.tsx
import Head from 'next/head';

// serviceType ONLY - the one field that MUST be manual
const SERVICE_TYPE_CONFIG: Record<string, string> = {
  'perm': 'Permanent Placement',
  'flexibility': 'Flexible Employment',
  'bchirim': 'Executive Recruitment',
  'on-site': 'On-site Team Management',
  'experts': 'Expert Staffing',
  'foreign-workers': 'Foreign Worker Employment',
  'payroll': 'Payroll Services',
  'c-level-candidates': 'C-Level Executive Placement',
  'manufacture': 'Manufacturing Management',
  'logistics': 'Logistics Outsourcing',
  'service-center': 'Call Center Outsourcing',
  'cleanrooms': 'Clean Room Maintenance',
  'technology-industry': 'Technology Industry Staffing',
  'service-plans': 'Internal Service Operations',
  'employers-solutions': 'Short-term Recruitment',
  'finance': 'Finance Recruitment',
  'life-science': 'Life Sciences Recruitment',
};

// Extract data from __NEXT_DATA__ pageProps based on template type
function extractServiceData(pageProps: any, slug: string) {
  // Template 1: Employers Solutions (14 pages)
  const inner = pageProps?.template?.innerEmployers;
  if (inner) {
    return {
      name: inner.topBanner?.title || '',
      description: stripHtml(inner.bannerCenterContent1?.content || inner.topBanner?.content || ''),
      faq: inner.accordionSection || [],
    };
  }
  // Template 2: Specializations (finance, life-science)
  const spec = pageProps?.template?.Specializations;
  if (spec) {
    return {
      name: spec.topBanner?.title || '',
      description: stripHtml(spec.bannerCenterContent1?.content || spec.topBanner?.content || ''),
      faq: [],
    };
  }
  // Template 3: Employers Manpower (foreign-workers)
  const emp = pageProps?.employers?.pages?.nodes?.[0]?.template?.employersManpower;
  if (emp) {
    return {
      name: emp.topBanner?.title || '',
      description: stripHtml(emp.imgRightWithTextLeft?.content || emp.topBanner?.content || ''),
      faq: emp.accordionSection || [],
    };
  }
  return null;
}

function stripHtml(html: string): string {
  return html.replace(/<[^>]+>/g, '').trim().slice(0, 300);
}

export default function ServiceSchema({ pageProps, slug }: { pageProps: any; slug: string }) {
  const serviceType = SERVICE_TYPE_CONFIG[slug];
  if (!serviceType) return null;

  const data = extractServiceData(pageProps, slug);
  if (!data) return null;

  const basePath = ['finance','life-science'].includes(slug) ? 'specializations'
    : ['foreign-workers'].includes(slug) ? 'employers' : 'employers-solutions';

  const schema = {
    '@context': 'https://schema.org/',
    '@type': 'Service',
    name: data.name,              // from __NEXT_DATA__ (automatic)
    serviceType,                   // from config (manual)
    description: data.description, // from __NEXT_DATA__ (automatic)
    url: `https://www.manpower.co.il/${basePath}/${slug}`,
    provider: { '@id': 'https://www.manpower.co.il/#organization' },
    areaServed: { '@type': 'Country', name: 'Israel' },
    audience: { '@type': 'Audience', audienceType: 'Employers' },
  };

  return (
    <Head>
      <script type="application/ld+json" dangerouslySetInnerHTML={{ __html: JSON.stringify(schema) }} />
    </Head>
  );
}
